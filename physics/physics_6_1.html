・加算、乗算

ベクトルの計算

\[ \begin{aligned} \vec{v}+s\vec{w} &=\sum_{i=3}^{3}\vec{\delta}_{i}v_i+s\sum_{i=3}^{3}\vec{\delta}_{i}w_i\\[6pt] &=\sum_{i=3}^{3}\vec{\delta}_{i}v_i+\sum_{i=3}^{3}\vec{\delta}_{i}sw_i\\[6pt] &=\sum_{i=3}^{3}\vec{\delta}_{i}(v_i+sw_i)\\[6pt] &=\vec{\delta}_{x}(v_x+sw_x)+\vec{\delta}_{y}(v_y+sw_y)+\vec{\delta}_{z}(v_z+sw_z) \end{aligned} \]

テンソルの計算

\[ \begin{aligned} \vec{\tau}+s\vec{\sigma} &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\vec{\delta}_{j}\tau_{ij}+s\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\vec{\delta}_{j}\sigma_{ij}\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\vec{\delta}_{j}\tau_{ij}+\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\vec{\delta}_{j}s\sigma_{ij}\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\vec{\delta}_{j}(\tau_{ij}+s\sigma_{ij})\\[6pt] &=\vec{\delta}_{x}\vec{\delta}_{x}(\tau_{xx}+s\sigma_{xx})+\vec{\delta}_{x}\vec{\delta}_{y}(\tau_{xy}+s\sigma_{xy})+\vec{\delta}_{x}\vec{\delta}_{z}(\tau_{xz}+s\sigma_{xz})\\[6pt] &\quad+\vec{\delta}_{y}\vec{\delta}_{x}(\tau_{yx}+s\sigma_{yx})+\vec{\delta}_{y}\vec{\delta}_{y}(\tau_{yy}+s\sigma_{yy})+\vec{\delta}_{y}\vec{\delta}_{z}(\tau_{yz}+s\sigma_{yz})\\[6pt] &\quad+\vec{\delta}_{z}\vec{\delta}_{x}(\tau_{zx}+s\sigma_{zx})+\vec{\delta}_{z}\vec{\delta}_{y}(\tau_{zy}+s\sigma_{zy})+\vec{\delta}_{z}\vec{\delta}_{z}(\tau_{zz}+s\sigma_{zz}) \end{aligned} \]

減算も同様に計算できます。


・微分作用素を含む場合

スカラーの微分

\[ \begin{aligned} \nabla s &=\sum_{i=0}^{3}\vec{\delta}_{i}\frac{\partial}{\partial x_i}s\\[6pt] &=\sum_{i=0}^{3}\vec{\delta}_{i}\frac{\partial s}{\partial x_i}\\[6pt] &=\vec{\delta}_{x}\frac{\partial s}{\partial x}+\vec{\delta}_{y}\frac{\partial s}{\partial y}+\vec{\delta}_{z}\frac{\partial s}{\partial z} \end{aligned} \]

ベクトルの微分

\[ \begin{aligned} \nabla\vec{v} &=\sum_{i=0}^{3}\vec{\delta}_{i}\frac{\partial}{\partial x_i}\sum_{j=0}^{3}\vec{\delta}_{j}v_j\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\frac{\partial}{\partial x_i}\vec{\delta}_{j}v_j\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\frac{\partial(\vec{\delta}_{j}v_j)}{\partial x_i}\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{i}\vec{\delta}_{j}\frac{\partial v_j}{\partial x_i}\\[6pt] &=\vec{\delta}_{x}\vec{\delta}_{x}\frac{\partial v_x}{\partial x}+\vec{\delta}_{x}\vec{\delta}_{y}\frac{\partial v_y}{\partial x}+\vec{\delta}_{x}\vec{\delta}_{z}\frac{\partial v_z}{\partial x}\\[6pt] &\quad+\vec{\delta}_{y}\vec{\delta}_{x}\frac{\partial v_x}{\partial y}+\vec{\delta}_{y}\vec{\delta}_{y}\frac{\partial v_y}{\partial y}+\vec{\delta}_{y}\vec{\delta}_{z}\frac{\partial v_z}{\partial y}\\[6pt] &\quad+\vec{\delta}_{z}\vec{\delta}_{x}\frac{\partial v_x}{\partial z}+\vec{\delta}_{z}\vec{\delta}_{y}\frac{\partial v_y}{\partial z}+\vec{\delta}_{z}\vec{\delta}_{z}\frac{\partial v_z}{\partial z} \end{aligned} \]

ベクトルのラプラシアン

\[ \begin{aligned} \Delta\vec{v} &=\sum_{i=0}^{3}\frac{\partial^2}{\partial x_i^2}\sum_{j=0}^{3}\vec{\delta}_{j}v_j\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\frac{\partial^2}{\partial x_i^2}\vec{\delta}_{j}v_j\\[6pt] &=\sum_{i=0}^{3}\sum_{j=0}^{3}\vec{\delta}_{j}\frac{\partial^2 v_j}{\partial x_i^2}\\[6pt] &=\sum_{j=0}^{3}\vec{\delta}_{j}\left(\frac{\partial^2 v_j}{\partial x^2}+\frac{\partial^2 v_j}{\partial y^2}+\frac{\partial^2 v_j}{\partial z^2}\right)\\[6pt] &=\vec{\delta}_{x}\left(\frac{\partial^2 v_x}{\partial x^2}+\frac{\partial^2 v_x}{\partial y^2}+\frac{\partial^2 v_x}{\partial z^2}\right)\\[6pt] &\quad+\vec{\delta}_{y}\left(\frac{\partial^2 v_y}{\partial x^2}+\frac{\partial^2 v_y}{\partial y^2}+\frac{\partial^2 v_y}{\partial z^2}\right)\\[6pt] &\quad+\vec{\delta}_{z}\left(\frac{\partial^2 v_z}{\partial x^2}+\frac{\partial^2 v_z}{\partial y^2}+\frac{\partial^2 v_z}{\partial z^2}\right) \end{aligned} \]
